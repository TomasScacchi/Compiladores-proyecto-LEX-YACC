CREATE TABLE empleados (
    legajo INT,
    nombre VARCHAR(30),
    apellido VARCHAR(30),
    salario DECIMAL(10,2),
    antiguedad INT
);

SELECT nombre, apellido, legajo FROM empleados;
SELECT COUNT(legajo), MAX(salario), MIN(antiguedad) FROM empleados;

SELECT nombre, apellido
FROM empleados
WHERE (salario > 50000.00 AND antiguedad >= 10);
